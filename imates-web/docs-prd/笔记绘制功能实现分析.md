# 笔记绘制功能实现分析

## 一、功能概述

笔记绘制功能是PDF查看器的核心功能之一，允许用户在PDF页面上进行绘制、标注、截图等操作。系统提供了两套实现方案：

1. **PDF页面标注**：基于Fabric.js在PDF页面上直接绘制
2. **独立画板**：使用原生Canvas API实现的独立绘制组件

### 2.2 核心组件职责

#### 2.2.1 UnifiedToolbar.vue
- **职责**：统一工具栏组件，提供工具选择和配置界面
- **功能**：
  - 工具切换（签字笔、荧光笔、橡皮擦、截图等）
  - 颜色选择
  - 粗细/大小配置
  - 撤销/重做等操作按钮

#### 2.2.2 PdfPage.vue
- **职责**：PDF页面组件，负责PDF渲染和标注层集成
- **功能**：
  - PDF内容渲染（使用PDF.js）
  - Fabric.js Canvas初始化
  - 标注数据加载和保存
  - 触摸手势处理（双指缩放/滑动）

#### 2.2.3 FabricCanvasServiceEnhanced.ts
- **职责**：Fabric.js服务层封装，提供框架无关的Canvas操作
- **功能**：
  - Canvas初始化和配置
  - 工具模式切换（pen/highlighter/eraser/screenshot）
  - 标注对象的增删改查
  - 截图功能实现
  - 事件监听和回调

#### 2.2.4 pdfViewerStore.ts
- **职责**：状态管理中心，管理PDF和标注数据
- **功能**：
  - PDF文档状态管理
  - 标注数据内存管理
  - 自动保存（防抖机制）
  - IndexedDB持久化

#### 2.2.5 DrawingBoard.vue
- **职责**：独立画板组件，提供完整的绘制功能
- **功能**：
  - 原生Canvas绘制（路径、矩形、圆形、直线、三角形、文本）
  - 对象选择、移动、删除
  - 撤销/重做
  - 缩放和拖拽
  - 触摸手势支持

```typescript
// FabricCanvasServiceEnhanced.ts - setEraserMode
private setEraserMode(): void {
  // 第1步：禁用绘制模式
  this.fabricCanvas.isDrawingMode = false
  this.fabricCanvas.selection = false
  
  // 第2步：设置橡皮擦事件监听
  this.setupEraserEvents()
}

// 橡皮擦事件处理
private setupEraserEvents(): void {
  const handleMouseDown = (event: any) => {
    this.isErasing = true
    const pointer = this.fabricCanvas?.getPointer(event.e)
    if (pointer) {
      this.checkAndDeleteObjects(pointer)
    }
  }
  
  const handleMouseMove = (event: any) => {
    if (this.isErasing) {
      const pointer = this.fabricCanvas?.getPointer(event.e)
      if (pointer) {
        this.checkAndDeleteObjects(pointer)
      }
    }
  }
  
  // 监听鼠标事件
  this.fabricCanvas.on('mouse:down', handleMouseDown)
  this.fabricCanvas.on('mouse:move', handleMouseMove)
  this.fabricCanvas.on('mouse:up', () => {
    this.isErasing = false
  })
}

// 检查并删除与橡皮擦相交的对象
private checkAndDeleteObjects(pointer: { x: number; y: number }): void {
  const eraserSize = this.config.eraserSize
  const eraserRadius = eraserSize / 2
  const objects = this.fabricCanvas.getObjects()
  
  objects.forEach((obj) => {
    if (obj.type === 'path') {
      const bounds = obj.getBoundingRect()
      
      // 计算最近点距离
      const closestX = Math.max(bounds.left, Math.min(pointer.x, bounds.left + bounds.width))
      const closestY = Math.max(bounds.top, Math.min(pointer.y, bounds.top + bounds.height))
      
      const distanceX = pointer.x - closestX
      const distanceY = pointer.y - closestY
      const distance = Math.sqrt(distanceX * distanceX + distanceY * distanceY)
      
      // 如果距离小于橡皮擦半径，删除对象
      if (distance < eraserRadius) {
        this.fabricCanvas?.remove(obj)
      }
    }
  })
  
  this.fabricCanvas.renderAll()
}
```

```typescript
// FabricCanvasServiceEnhanced.ts - setScreenshotMode
private setScreenshotMode(): void {
  // 第1步：禁用绘制模式
  this.fabricCanvas.isDrawingMode = false
  this.fabricCanvas.selection = false
  
  // 第2步：设置截图事件监听
  this.setupScreenshotEvents()
}

// 截图事件处理（矩形截图）
private setupScreenshotEvents(): void {
  const handleMouseDown = (event: any) => {
    const pointer = this.fabricCanvas?.getPointer(event.e)
    if (!pointer) return
    
    // 第1步：记录开始点
    this.screenshotState.isDrawing = true
    this.screenshotState.startPoint = { x: pointer.x, y: pointer.y }
    
    // 第2步：创建矩形选区
    const rect = new Rect({
      left: pointer.x,
      top: pointer.y,
      width: 0,
      height: 0,
      fill: this.config.screenshotFillColor,
      stroke: this.config.screenshotStrokeColor,
      strokeWidth: this.config.screenshotStrokeWidth,
      selectable: false,
      evented: false,
    })
    
    this.fabricCanvas?.add(rect)
    this.screenshotState.currentRect = rect
  }
  
  const handleMouseMove = (event: any) => {
    if (this.screenshotState.isDrawing && this.screenshotState.currentRect) {
      const pointer = this.fabricCanvas?.getPointer(event.e)
      const startPoint = this.screenshotState.startPoint
      
      if (pointer && startPoint) {
        // 更新矩形尺寸
        const width = pointer.x - startPoint.x
        const height = pointer.y - startPoint.y
        
        this.screenshotState.currentRect.set({
          left: width > 0 ? startPoint.x : pointer.x,
          top: height > 0 ? startPoint.y : pointer.y,
          width: Math.abs(width),
          height: Math.abs(height),
        })
        
        this.fabricCanvas?.renderAll()
      }
    }
  }
  
  const handleMouseUp = () => {
    if (this.screenshotState.isDrawing && this.screenshotState.currentRect) {
      // 捕获截图区域
      this.captureScreenshotArea(this.screenshotState.currentRect)
    }
  }
}

// 捕获截图
private async captureScreenshotArea(shape: FabricObject): Promise<void> {
  // 第1步：获取选区边界
  shape.setCoords()
  const boundingRect = shape.getBoundingRect()
  
  // 第2步：创建临时Canvas
  const tempCanvas = document.createElement('canvas')
  tempCanvas.width = Math.max(1, Math.round(boundingRect.width))
  tempCanvas.height = Math.max(1, Math.round(boundingRect.height))
  const tempCtx = tempCanvas.getContext('2d')
  
  // 第3步：绘制PDF层
  tempCtx.drawImage(
    this.pdfCanvas,
    Math.round(boundingRect.left),
    Math.round(boundingRect.top),
    boundingRect.width,
    boundingRect.height,
    0,
    0,
    boundingRect.width,
    boundingRect.height
  )
  
  // 第4步：绘制Fabric层（临时隐藏选区形状）
  const shapeVisible = shape.visible
  shape.set({ visible: false })
  this.fabricCanvas.renderAll()
  
  const fabricElement = this.fabricCanvas.getElement()
  if (fabricElement) {
    tempCtx.drawImage(
      fabricElement,
      Math.round(boundingRect.left),
      Math.round(boundingRect.top),
      boundingRect.width,
      boundingRect.height,
      0,
      0,
      boundingRect.width,
      boundingRect.height
    )
  }
  
  // 第5步：恢复选区可见性
  shape.set({ visible: shapeVisible })
  this.fabricCanvas.renderAll()
  
  // 第6步：转换为Blob并触发回调
  tempCanvas.toBlob((blob) => {
    if (blob) {
      this.events.onScreenshotCaptured?.(blob)
      this.fabricCanvas?.remove(shape)
      this.fabricCanvas?.renderAll()
      this.resetScreenshotState()
    }
  }, 'image/jpeg', 0.4)
}
```

```typescript
// pdfViewerStore.ts - updateAnnotations
updateAnnotations(pageNum: number, fabricJson: object[]) {
  // 第1步：更新内存中的数据
  if (fabricJson && fabricJson.length > 0) {
    this.allAnnotations[pageNum] = fabricJson
  } else {
    delete this.allAnnotations[pageNum]
  }
  
  // 第2步：更新兼容性Map
  const pageId = `page-${pageNum}`
  if (fabricJson && fabricJson.length > 0) {
    const noteData: NoteData[] = fabricJson.map((obj: object, index: number) => ({
      id: `${pageId}-${index}`,
      pageId,
      type: 'fabric',
      content: obj,
      config: this.drawingConfig,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    }))
    this.notes.set(pageId, noteData)
  } else {
    this.notes.delete(pageId)
  }
  
  // 第3步：触发防抖保存
  this.debouncedSave()
}

// 防抖保存
debouncedSave() {
  // 清除之前的定时器
  if (saveDebounceTimer) {
    clearTimeout(saveDebounceTimer)
    saveDebounceTimer = null
  }
  
  // 设置新的定时器（1秒后保存）
  saveDebounceTimer = setTimeout(async () => {
    await this.autoSaveAnnotations()
  }, 1000)
}

// 自动保存到IndexedDB
async autoSaveAnnotations() {
  if (!this.currentFileId || !this.currentResourceId) {
    return
  }
  
  if (this.isSaving) {
    return
  }
  
  try {
    this.isSaving = true
    this.saveError = null
    await this.saveAnnotationsToLocalFile()
    this.lastSaveTime = Date.now()
  } catch (error) {
    this.saveError = error instanceof Error ? error.message : '保存失败'
    console.error('自动保存笔记失败:', error)
  } finally {
    this.isSaving = false
  }
}
```

### 3.5 触摸手势处理

#### 3.5.1 双指缩放

```typescript
// PdfPage.vue - handleTouchMove
const handleTouchMove = (event: TouchEvent) => {
  if (event.touches.length !== 2) return
  
  const touch1 = event.touches[0]
  const touch2 = event.touches[1]
  const currentDistance = getDistance(touch1, touch2)
  const initialDistance = touchState.value.initialDistance
  
  // 判断是缩放还是滑动
  const distanceChange = Math.abs(currentDistance - initialDistance)
  const deltaY = Math.abs(currentY - touchState.value.startY)
  
  if (distanceChange > 15 && distanceChange > deltaY * 0.5) {
    // 缩放模式
    touchState.value.isZooming = true
    const scaleRatio = currentDistance / initialDistance
    const newScale = touchState.value.initialScale * scaleRatio
    const clampedScale = Math.max(0.5, Math.min(3.0, newScale))
    store.setScale(clampedScale)
  } else {
    // 滑动模式
    const scrollContainer = getScrollContainer()
    if (scrollContainer) {
      scrollContainer.scrollTop -= (currentY - touchState.value.lastY)
    }
  }
}
```

```typescript
        height: Math.abs(coords.y - startPoint.value.y),
      }
      render()
      break
      
    // ... 其他工具
  }
}

// handleMouseUp
const handleMouseUp = () => {
  if (tempObject.value && isDrawing.value) {
    // 添加临时对象到列表
    objects.value.push(tempObject.value)
    tempObject.value = null
    saveState()
    emit('content-change')
  }
  
  isDrawing.value = false
  currentPath.value = []
  startPoint.value = null
}
```

### 4.2 渲染机制

```typescript
// DrawingBoard.vue - render
const render = () => {
  if (!ctx || !canvasRef.value) return
  
  // 第1步：清空画布
  ctx.clearRect(0, 0, canvasRef.value.width, canvasRef.value.height)
  
  // 第2步：绘制所有对象
  objects.value.forEach((obj, index) => {
    ctx.save()
    
    // 设置透明度（橡皮擦悬停效果）
    if (index === hoveredObject.value && currentTool.value === 'eraser-draw') {
      ctx.globalAlpha = 0.5
    } else if (obj.opacity !== undefined) {
      ctx.globalAlpha = obj.opacity
    }
    
    // 绘制对象
    drawObject(obj)
    
    ctx.restore()
    
    // 如果对象被选中，绘制高亮边框
    if (selectedObjects.value.has(index)) {
      drawObjectHighlight(obj)
    }
  })
  
  // 第3步：绘制临时对象（正在绘制中）
  if (tempObject.value) {
    ctx.save()
    drawObject(tempObject.value)
    ctx.restore()
  }
  
  // 第4步：绘制选框
  if (selectionBox.value) {
    drawSelectionBox(selectionBox.value)
  }
}
```

### 4.3 撤销/重做机制

```typescript
// DrawingBoard.vue - 历史记录管理
const history = ref<DrawObject[][]>([[]])
const historyIndex = ref(0)

// 保存状态
const saveState = () => {
  // 如果当前不在历史记录末尾，删除后面的记录
  if (historyIndex.value < history.value.length - 1) {
    history.value = history.value.slice(0, historyIndex.value + 1)
  }
  
  // 保存当前状态（深拷贝）
  history.value.push(JSON.parse(JSON.stringify(objects.value)))
  historyIndex.value = history.value.length - 1
  
  // 限制历史记录数量
  if (history.value.length > 20) {
    history.value.shift()
    historyIndex.value--
  }
  
  emit('content-change')
}

// 撤销
const undo = () => {
  if (historyIndex.value > 0) {
    historyIndex.value--
    objects.value = JSON.parse(JSON.stringify(history.value[historyIndex.value]))
    render()
    emit('content-change')
  }
}

// 重做
const redo = () => {
  if (historyIndex.value < history.value.length - 1) {
    historyIndex.value++
    objects.value = JSON.parse(JSON.stringify(history.value[historyIndex.value]))
    render()
    emit('content-change')
  }
}
```

```typescript
// DrawingBoard.vue - 触摸事件处理
const handleTouchStart = (e: TouchEvent) => {
  e.preventDefault()
  
  if (e.touches.length === 2) {
    // 双指手势：缩放或平移
    isTwoFingerGesture.value = true
    initialTouchDistance.value = getTouchDistance(e.touches[0], e.touches[1])
    initialTouchScale.value = zoomLevel.value
    
    const center = getTouchCenter(e.touches[0], e.touches[1])
    initialTouchCenterX.value = center.x
    initialTouchCenterY.value = center.y
    initialTouchTranslateX.value = canvasOffset.value.x
    initialTouchTranslateY.value = canvasOffset.value.y
  } else if (e.touches.length === 1) {
    // 单指触摸：延迟判断（可能是双指操作）
    pendingSingleTouch.value = true
    pendingTouchX.value = e.touches[0].clientX
    pendingTouchY.value = e.touches[0].clientY
    
    singleTouchTimer = window.setTimeout(() => {
      if (pendingSingleTouch.value && !isTwoFingerGesture.value) {
        // 转换为鼠标事件
        const mouseEvent = new MouseEvent('mousedown', {
          clientX: pendingTouchX.value,
          clientY: pendingTouchY.value,
        })
        handleMouseDown(mouseEvent)
        pendingSingleTouch.value = false
      }
      singleTouchTimer = null
    }, 80)
  }
}

const handleTouchMove = (e: TouchEvent) => {
  e.preventDefault()
  
  if (e.touches.length === 2) {
    // 双指缩放/平移
    const currentDistance = getTouchDistance(e.touches[0], e.touches[1])
    const currentCenter = getTouchCenter(e.touches[0], e.touches[1])
    
    // 计算缩放
    const scaleChange = currentDistance / initialTouchDistance.value
    const newScale = initialTouchScale.value * scaleChange
    const clampedScale = Math.min(Math.max(newScale, 0.1), 3)
    
    // 计算平移
    const centerDeltaX = currentCenter.x - initialTouchCenterX.value
    const centerDeltaY = currentCenter.y - initialTouchCenterY.value
    
    // 更新缩放和偏移
    zoomLevel.value = clampedScale
    canvasOffset.value = {
      x: initialTouchTranslateX.value + centerDeltaX,
      y: initialTouchTranslateY.value + centerDeltaY,
    }
  } else if (e.touches.length === 1 && !isTwoFingerGesture.value) {
    // 单指绘制
    const touch = e.touches[0]
    const mouseEvent = new MouseEvent('mousemove', {
      clientX: touch.clientX,
      clientY: touch.clientY,
    })
    handleMouseMove(mouseEvent)
  }
}
```
